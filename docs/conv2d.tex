\section{Overview of 2D Convolution}

A 2D convolution layer applies learnable filters to an input tensor to produce feature maps, commonly used in convolutional neural networks (CNNs). This manual provides a step-by-step guide to implement a \texttt{Conv2d} layer, including forward and backward passes, parameter initialization, and support for dilation (atrous convolution). The implementation uses the \texttt{im2col} algorithm for efficiency.

\subsection{Key Parameters}
\begin{itemize}
    \item \textbf{Input Tensor}: Shape $[B, C\_{\text{in}}, H, W]$ (denoted $X$).
    \item \textbf{Output Tensor}: Shape $[B, C\_{\text{out}}, H\_{\text{out}}, W\_{\text{out}}]$ (denoted $Y$).
    \item \textbf{Weights (Kernels)}: Shape $[C\_{\text{out}}, C\_{\text{in}}, k\_h, k\_w]$ (denoted $W$).
    \item \textbf{Biases} (optional): Shape $[C\_{\text{out}}]$ (denoted $b$).
    \item \textbf{Kernel Size}: $(k\_h, k\_w)$, height and width of each filter.
    \item \textbf{Stride}: $(s\_h, s\_w)$, step size for sliding the kernel.
    \item \textbf{Padding}: Options: ``valid'' (no padding), ``same'' (maintain input size), or custom $(p\_{\text{left}}, p\_{\text{right}}, p\_{\text{top}}, p\_{\text{bottom}})$.
    \item \textbf{Padding Mode}: ``zero'' (zero-padding).
    \item \textbf{Dilation}: $(d\_h, d\_w)$, spacing between kernel elements (default $(1, 1)$).
    \item \textbf{Activation Function}: Applied element-wise (e.g., ReLU).
    \item \textbf{Trainable}: Boolean for parameter updates.
    \item \textbf{Bias}: Boolean for including biases.
\end{itemize}

\subsection{Assumptions}
\begin{itemize}
    \item Tensors are 4D with dimensions $[B, C, H, W]$.
    \item Dilation defaults to $(d\_h, d\_w) = (1, 1)$ for standard convolution.
    \item Convolution is implemented using \texttt{im2col}.
\end{itemize}

\section{Mathematical Formulation}

\subsection{Forward Pass}
The forward pass computes $Y$ by convolving $X$ with $W$, adding biases $b$ (if applicable), and applying an activation function $\sigma$. For $Y[b, oc, oh, ow]$:

\begin{equation}
Y[b, oc, oh, ow] = \sum\_{ic=0}^{C\_{\text{in}}-1} \sum\_{kh=0}^{k\_h-1} \sum\_{kw=0}^{k\_w-1} X[b, ic, ih, iw] \cdot W[oc, ic, kh, kw] + b[oc]
\end{equation}

Where:
\begin{itemize}
    \item $ih = oh \cdot s\_h + kh \cdot d\_h - p\_{\text{top}}$
    \item $iw = ow \cdot s\_w + kw \cdot d\_w - p\_{\text{left}}$
\end{itemize}

The output is activated:
\begin{equation}
Y[b, oc, oh, ow] = \sigma(Y[b, oc, oh, ow])
\end{equation}

\subsubsection{Output Dimensions}
\begin{align}
H\_{\text{out}} &= \left\lfloor \frac{H + p\_{\text{top}} + p\_{\text{bottom}} - d\_h \cdot (k\_h - 1) - 1}{s\_h} \right\rfloor + 1 \\
W\_{\text{out}} &= \left\lfloor \frac{W + p\_{\text{left}} + p\_{\text{right}} - d\_w \cdot (k\_w - 1) - 1}{s\_w} \right\rfloor + 1
\end{align}

For ``same'' padding:
\begin{align}
p\_{h\_{\text{total}}} &= \max(0, (H\_{\text{out}} - 1) \cdot s\_h + d\_h \cdot (k\_h - 1) + 1 - H) \\
p\_{w\_{\text{total}}} &= \max(0, (W\_{\text{out}} - 1) \cdot s\_w + d\_w \cdot (k\_w - 1) + 1 - W) \\
p\_{\text{top}} &= \left\lfloor \frac{p\_{h\_{\text{total}}}}{2} \right\rfloor, \quad p\_{\text{bottom}} = p\_{h\_{\text{total}}} - p\_{\text{top}} \\
p\_{\text{left}} &= \left\lfloor \frac{p\_{w\_{\text{total}}}}{2} \right\rfloor, \quad p\_{\text{right}} = p\_{w\_{\text{total}}} - p\_{\text{left}}
\end{align}
Where $H\_{\text{out}} = \lceil H / s\_h \rceil$, $W\_{\text{out}} = \lceil W / s\_w \rceil$.

\subsection{Backward Pass}
The backward pass computes gradients $\frac{\partial L}{\partial X}$, $\frac{\partial L}{\partial W}$, and $\frac{\partial L}{\partial b}$ given $\frac{\partial L}{\partial Y}$ (denoted $\textbf{grad\_out}$).

\subsubsection{Gradient w.r.t. Input}
\begin{equation}
\frac{\partial L}{\partial X[b, ic, ih, iw]} = \sum\_{oc=0}^{C\_{\text{out}}-1} \sum\_{kh=0}^{k\_h-1} \sum\_{kw=0}^{k\_w-1} \frac{\partial L}{\partial Y'[b, oc, oh, ow]} \cdot W[oc, ic, kh, kw]
\end{equation}
Where:
\begin{itemize}
    \item $\frac{\partial L}{\partial Y'} = \sigma'(\text{pre\_activation}) \cdot \frac{\partial L}{\partial Y}$
    \item $oh = \lfloor (ih + p\_{\text{top}} - kh \cdot d\_h) / s\_h \rfloor$
    \item $ow = \lfloor (iw + p\_{\text{left}} - kw \cdot d\_w) / s\_w \rfloor$
\end{itemize}

\subsubsection{Gradient w.r.t. Weights}
\begin{equation}
\frac{\partial L}{\partial W[oc, ic, kh, kw]} = \sum\_{b=0}^{B-1} \sum\_{oh=0}^{H\_{\text{out}}-1} \sum\_{ow=0}^{W\_{\text{out}}-1} \frac{\partial L}{\partial Y'[b, oc, oh, ow]} \cdot X[b, ic, ih, iw]
\end{equation}
Where:
\begin{itemize}
    \item $ih = oh \cdot s\_h + kh \cdot d\_h - p\_{\text{top}}$
    \item $iw = ow \cdot s\_w + kw \cdot d\_w - p\_{\text{left}}$
\end{itemize}

\subsubsection{Gradient w.r.t. Biases}
\begin{equation}
\frac{\partial L}{\partial b[oc]} = \sum\_{b=0}^{B-1} \sum\_{oh=0}^{H\_{\text{out}}-1} \sum\_{ow=0}^{W\_{\text{out}}-1} \frac{\partial L}{\partial Y'[b, oc, oh, ow]}
\end{equation}

\subsection{Parameter Update}
Using an optimizer (e.g., SGD):
\begin{align}
W &\gets W - \eta \cdot \frac{\partial L}{\partial W} \\
b &\gets b - \eta \cdot \frac{\partial L}{\partial b}
\end{align}
Where $\eta$ is the learning rate.

\section{Implementation Details}

\subsection{Class Structure}
The \texttt{Conv2d} class includes:
\begin{itemize}
    \item \textbf{Attributes}: Channels, kernel size, \text{stride}, padding, dilation, weights ($W$), biases ($b$), gradients, input/output caches, activation function.
    \item \textbf{Methods}: Constructor, \texttt{forward}, \texttt{backward}, \texttt{update\_parameters}, \texttt{reset\_grads}, helpers (\texttt{pad\_input}, \texttt{init\_output}, \texttt{im2col}, \texttt{col2im\_add}).
\end{itemize}

\subsection{Constructor}
\begin{algorithm}
\caption{Conv2d Constructor}
\begin{algorithmic}[1]
\Function{Conv2d}{in\_channels, out\_channels, kernel\_size=$(3,3)$, \text{stride}=$(1,1)$, padding="valid", num\_padding=$(0,0,0,0)$, padding\_mode="zero", dilation=$(1,1)$, activator=null, layer\_name="Conv2D", trainable=true, bias=true}
    \If{in\_channels $\leq 0$ or out\_channels $\leq 0$}
        \State \textbf{throw} "Channels must be positive"
    \EndIf
    \If{kernel\_size[0] $\leq 0$ or kernel\_size[1] $\leq 0$}
        \State \textbf{throw} "Kernel size must be positive"
    \EndIf
    \If{\text{stride}[0] $\leq 0$ or \text{stride}[1] $\leq 0$}
        \State \textbf{throw} "Stride must be positive"
    \EndIf
    \If{dilation[0] $\leq 0$ or dilation[1] $\leq 0$}
        \State \textbf{throw} "Dilation must be positive"
    \EndIf
    \If{padding $\notin$ ["valid", "same", "none"]}
        \State \textbf{throw} "Invalid padding mode"
    \EndIf
    \If{padding\_mode $\neq$ "zero"}
        \State \textbf{throw} "Only zero-padding is supported"
    \EndIf
    \If{num\_padding[0,1,2,3] $< 0$}
        \State \textbf{throw} "Padding values must be non-negative"
    \EndIf
    \State this.in\_channels $\gets$ in\_channels
    \State this.out\_channels $\gets$ out\_channels
    \State this.kernel\_size $\gets$ kernel\_size
    \State this.\text{stride} $\gets$ \text{stride}
    \State this.padding $\gets$ padding
    \State this.num\_padding $\gets$ num\_padding
    \State this.padding\_mode $\gets$ padding\_mode
    \State this.dilation $\gets$ dilation
    \State this.layer\_name $\gets$ layer\_name or "Conv2D\_\{in\_channels\}x\{out\_channels\}"
    \State this.trainable $\gets$ trainable
    \State this.bias $\gets$ bias
    \If{activator is null}
        \State this.activator $\gets$ new ReLU()
    \Else
        \State this.activator $\gets$ activator
    \EndIf
    \State init\_params\_and\_grads()
\EndFunction
\end{algorithmic}
\end{algorithm}

\subsubsection{Parameter Initialization}
Use Xavier initialization:
\begin{equation}
\text{scale} = \sqrt{\frac{6}{\text{fan}\_{\text{in}} + \text{fan}\_{\text{out}}}}
\end{equation}
Where $\text{fan}\_{\text{in}} = C\_{\text{in}} \cdot k\_h \cdot k\_w$, $\text{fan}\_{\text{out}} = C\_{\text{out}} \cdot k\_h \cdot k\_w$.

\begin{algorithm}
\caption{Initialize Parameters and Gradients}
\begin{algorithmic}[1]
\Function{init\_params\_and\_grads}{}
    \State fan\_in $\gets$ in\_channels $\cdot$ kernel\_size[0] $\cdot$ kernel\_size[1]
    \State fan\_out $\gets$ out\_channels $\cdot$ kernel\_size[0] $\cdot$ kernel\_size[1]
    \State scale $\gets$ $\sqrt{6 / (\text{fan}\_{\text{in}} + \text{fan}\_{\text{out}})}$
    \State weights $\gets$ random\_uniform(-scale, scale, shape=[out\_channels, in\_channels, kernel\_size[0], kernel\_size[1]])
    \State grad\_weights $\gets$ zeros(shape=[out\_channels, in\_channels, kernel\_size[0], kernel\_size[1]])
    \If{bias}
        \State biases $\gets$ zeros(shape=[out\_channels])
        \State grad\_biases $\gets$ zeros(shape=[out\_channels])
    \Else
        \State biases $\gets$ empty\_tensor()
        \State grad\_biases $\gets$ empty\_tensor()
    \EndIf
\EndFunction
\end{algorithmic}
\end{algorithm}

\subsection{Padding}
\begin{algorithm}
\caption{Pad Input}
\begin{algorithmic}[1]
\Function{pad\_input}{X}
    \State $p\_{\text{left}}, p\_{\text{right}}, p\_{\text{top}}, p\_{\text{bottom}} \gets$ num\_padding
    \If{padding = "same"}
        \State $k\_h, k\_w \gets$ kernel\_size
        \State $s\_h, s\_w \gets$ \text{stride}
        \State $d\_h, d\_w \gets$ dilation
        \State $H, W \gets$ X.shape[2], X.shape[3]
        \State out\_h $\gets$ $\lceil H / s\_h \rceil$
        \State out\_w $\gets$ $\lceil W / s\_w \rceil$
        \State $p\_{h\_{\text{total}}} \gets$ $\max(0, (\text{out}\_h - 1) \cdot s\_h + d\_h \cdot (k\_h - 1) + 1 - H)$
        \State $p\_{w\_{\text{total}}} \gets$ $\max(0, (\text{out}\_w - 1) \cdot s\_w + d\_w \cdot (k\_w - 1) + 1 - W)$
        \State $p\_{\text{top}} \gets$ $\lfloor p\_{h\_{\text{total}}} / 2 \rfloor$
        \State $p\_{\text{bottom}} \gets$ $p\_{h\_{\text{total}}} - p\_{\text{top}}$
        \State $p\_{\text{left}} \gets$ $\lfloor p\_{w\_{\text{total}}} / 2 \rfloor$
        \State $p\_{\text{right}} \gets$ $p\_{w\_{\text{total}}} - p\_{\text{left}}$
    \EndIf
    \If{$p\_{\text{left}} = 0$ and $p\_{\text{right}} = 0$ and $p\_{\text{top}} = 0$ and $p\_{\text{bottom}} = 0$}
        \State \Return X
    \EndIf
    \State padded\_X $\gets$ zeros(shape=[X.shape[0], X.shape[1], $H + p\_{\text{top}} + p\_{\text{bottom}}, W + p\_{\text{left}} + p\_{\text{right}}$])
    \State padded\_X[:, :, $p\_{\text{top}}:H+p\_{\text{top}}, p\_{\text{left}}:W+p\_{\text{left}}$] $\gets$ X
    \State \Return padded\_X
\EndFunction
\end{algorithmic}
\end{algorithm}

\subsection{Output Dimensions}
\begin{algorithm}
\caption{Initialize Output}
\begin{algorithmic}[1]
\Function{init\_output}{X}
    \State $B, C, H, W \gets$ X.shape
    \State $k\_h, k\_w \gets$ kernel\_size
    \State $s\_h, s\_w \gets$ \text{stride}
    \State $d\_h, d\_w \gets$ dilation
    \State $p\_{\text{top}}, p\_{\text{bottom}}, p\_{\text{left}}, p\_{\text{right}} \gets$ num\_padding
    \If{padding in ["valid", "none"]}
        \State out\_h $\gets$ $\lfloor (H + p\_{\text{top}} + p\_{\text{bottom}} - d\_h \cdot (k\_h - 1) - 1) / s\_h \rfloor + 1$
        \State out\_w $\gets$ $\lfloor (W + p\_{\text{left}} + p\_{\text{right}} - d\_w \cdot (k\_w - 1) - 1) / s\_w \rfloor + 1$
        \If{$(H + p\_{\text{top}} + p\_{\text{bottom}} - d\_h \cdot (k\_h - 1) - 1) \mod s\_h \neq 0$ or $(W + p\_{\text{left}} + p\_{\text{right}} - d\_w \cdot (k\_w - 1) - 1) \mod s\_w \neq 0$}
            \State \textbf{throw} "Non-integer output dimensions"
        \EndIf
    \ElsIf{padding = "same"}
        \State out\_h $\gets$ $\lceil H / s\_h \rceil$
        \State out\_w $\gets$ $\lceil W / s\_w \rceil$
    \EndIf
    \If{out\_h $\leq 0$ or out\_w $\leq 0$}
        \State \textbf{throw} "Invalid output dimensions"
    \EndIf
    \State output $\gets$ zeros(shape=[B, out\_channels, out\_h, out\_w])
    \State \Return out\_h, out\_w
\EndFunction
\end{algorithmic}
\end{algorithm}

\subsection{Im2col Transformation}
\begin{algorithm}
\caption{Im2col Transformation}
\begin{algorithmic}[1]
\Function{im2col}{input}
    \State $B, C, H, W \gets$ input.shape
    \State $k\_h, k\_w \gets$ kernel\_size
    \State $s\_h, s\_w \gets$ \text{stride}
    \State $d\_h, d\_w \gets$ dilation
    \State out\_h $\gets$ $\lfloor (H - d\_h \cdot (k\_h - 1) - 1) / s\_h \rfloor + 1$
    \State out\_w $\gets$ $\lfloor (W - d\_w \cdot (k\_w - 1) - 1) / s\_w \rfloor + 1$
    \If{out\_h $\leq 0$ or out\_w $\leq 0$}
        \State \textbf{throw} "Invalid output dimensions in im2col"
    \EndIf
    \State col\_height $\gets$ $k\_h \cdot k\_w \cdot$ in\_channels
    \State col\_width $\gets$ $B \cdot$ out\_h $\cdot$ out\_w
    \State col\_matrix $\gets$ zeros(shape=[col\_height, col\_width])
    \State col\_idx $\gets$ 0
    \For{b in 0 to $B-1$}
        \For{oh in 0 to out\_h$-1$}
            \For{ow in 0 to out\_w$-1$}
                \State row\_idx $\gets$ 0
                \For{ic in 0 to in\_channels$-1$}
                    \For{kh in 0 to $k\_h-1$}
                        \For{kw in 0 to $k\_w-1$}
                            \State ih $\gets$ oh $\cdot s\_h + kh \cdot d\_h$
                            \State iw $\gets$ ow $\cdot s\_w + kw \cdot d\_w$
                            \If{ih $< H$ and iw $< W$}
                                \State col\_matrix[row\_idx, col\_idx] $\gets$ input[b, ic, ih, iw]
                            \EndIf
                            \State row\_idx $\gets$ row\_idx + 1
                        \EndFor
                    \EndFor
                \EndFor
                \State col\_idx $\gets$ col\_idx + 1
            \EndFor
        \EndFor
    \EndFor
    \State \Return col\_matrix
\EndFunction
\end{algorithmic}
\end{algorithm}

\subsection{Forward Pass}
\begin{algorithm}
\caption{Forward Pass}
\begin{algorithmic}[1]
\Function{forward}{X}
    \State in\_cache $\gets$ X
    \State $B, C, H, W \gets$ X.shape
    \If{$C \neq$ in\_channels}
        \State \textbf{throw} "Input channels mismatch"
    \EndIf
    \If{$H < $ kernel\_size[0] or $W < $ kernel\_size[1]}
        \State \textbf{throw} "Input dimensions too small"
    \EndIf
    \State input\_padded $\gets$ pad\_input(X)
    \State out\_h, out\_w $\gets$ init\_output(X)
    \State col\_matrix $\gets$ im2col(input\_padded)
    \State weight\_matrix $\gets$ weights.reshape([out\_channels, $k\_h \cdot k\_w \cdot$ in\_channels])
    \State result $\gets$ weight\_matrix $\cdot$ col\_matrix
    \If{bias}
        \State bias\_broadcasted $\gets$ biases.reshape([out\_channels, 1]).broadcast([1, $B \cdot$ out\_h $\cdot$ out\_w])
        \State result $\gets$ result + bias\_broadcasted
    \EndIf
    \State pre\_activation $\gets$ result.reshape([B, out\_channels, out\_h, out\_w])
    \State output $\gets$ activator.forward(pre\_activation)
    \State \Return output
\EndFunction
\end{algorithmic}
\end{algorithm}

\subsection{Col2im Transformation}
\begin{algorithm}
\caption{Col2im Add}
\begin{algorithmic}[1]
\Function{col2im\_add}{col\_matrix, grad\_input}
    \State $B, C, H, W \gets$ grad\_input.shape
    \State $k\_h, k\_w \gets$ kernel\_size
    \State $s\_h, s\_w \gets$ \text{stride}
    \State $d\_h, d\_w \gets$ dilation
    \State col\_idx $\gets$ 0
    \For{b in 0 to $B-1$}
        \For{oh in 0 to out\_h$-1$}
            \For{ow in 0 to out\_w$-1$}
                \State row\_idx $\gets$ 0
                \For{ic in 0 to in\_channels$-1$}
                    \For{kh in 0 to $k\_h-1$}
                        \For{kw in 0 to $k\_w-1$}
                            \State ih $\gets$ oh $\cdot s\_h + kh \cdot d\_h$
                            \State iw $\gets$ ow $\cdot s\_w + kw \cdot d\_w$
                            \If{ih $< H$ and iw $< W$}
                                \State grad\_input[b, ic, ih, iw] $\gets$ grad\_input[b, ic, ih, iw] + col\_matrix[row\_idx, col\_idx]
                            \EndIf
                            \State row\_idx $\gets$ row\_idx + 1
                        \EndFor
                    \EndFor
                \EndFor
                \State col\_idx $\gets$ col\_idx + 1
            \EndFor
        \EndFor
    \EndFor
\EndFunction
\end{algorithmic}
\end{algorithm}

\subsection{Backward Pass}
\begin{algorithm}
\caption{Backward Pass}
\begin{algorithmic}[1]
\Function{backward}{grad\_out}
    \If{grad\_out.shape $\neq$ [B, out\_channels, out\_h, out\_w]}
        \State \textbf{throw} "Gradient output dimensions mismatch"
    \EndIf
    \State grad\_input $\gets$ zeros(shape=in\_cache.shape)
    \If{trainable}
        \State grad\_weights $\gets$ zeros(shape=weights.shape)
        \If{bias}
            \State grad\_biases $\gets$ zeros(shape=biases.shape)
        \EndIf
    \EndIf
    \State input\_padded $\gets$ pad\_input(in\_cache)
    \State grad\_pre\_activation $\gets$ activator.backward(grad\_out)
    \State grad\_out\_matrix $\gets$ grad\_pre\_activation.reshape([out\_channels, $B \cdot$ out\_h $\cdot$ out\_w])
    \If{trainable}
        \State col\_matrix $\gets$ im2col(input\_padded)
        \State weight\_grad\_matrix $\gets$ grad\_out\_matrix $\cdot$ col\_matrix.transpose()
        \State grad\_weights $\gets$ weight\_grad\_matrix.data().reshape([out\_channels, in\_channels, $k\_h, k\_w$])
        \If{bias}
            \State grad\_biases $\gets$ grad\_out\_matrix.sum(axis=1)
        \EndIf
    \EndIf
    \State weight\_matrix\_weights.reshape([$k\_h \cdot k\_w \cdot$ in\_channelsgrad\_out\_matrix, out\_channels])
    \State grad\_input\_col $\gets$ grad\_out\_matrixweight\_matrix\_T * grad\_out\_matrix
    \State col2im\_add(grad\_input\_col, grad\_input)
    \State \Return grad\_input
\EndFunction
\end{algorithmic}
\end{algorithm}

\subsection{Parameter Update}
\begin{algorithm}
\caption{Update Parameters}
\begin{algorithmic}[1]
\Function{update\_parameters}{optimizer, learning\_rate}
    \If{not trainable}
        \State \Return
    \EndIf
    \If{weights.shape $\neq$ grad\_weights.shape}
        \State \textbf{throw} "Weight and gradient dimension mismatch"
    \EndIf
    \If{bias and biases.shape $\neq$ grad\_biases.shape}
        \State \textbf{throw} "Bias and gradient dimension mismatch"
    \EndIf
    \State optimizer.update(this, learning\_rate)
\EndFunction
\end{algorithmic}
\end{algorithm}

\subsection{Reset Gradients}
\begin{algorithm}
\caption{Reset Gradients}
\begin{algorithmic}[1]
\Function{reset\_grads}{}
    \If{trainable}
        \State grad\_weights $\gets$ zeros(shape=grad\_weights.shape)
        \If{bias}
            \State grad\_biases $\gets$ zeros(shape=grad\_biases.shape)
        \EndIf
    \EndIf
\EndFunction
\end{algorithmic}
\end{algorithm}

\section{Example Usage}
\begin{verbatim}
conv = Conv2d(
    in\_channels=3, out\_channels=64, kernel\_size=(3,3),
    \text{stride}=(1,1), padding="same", dilation=(2,2),
    activator=True, bias=True, trainable=True
)
X = random\_tensor(shape=[batch\_size, 3, 28, 28])
Y = conv.forward(X)
grad\_out = random\_tensor(shape=Y.shape)
grad\_input = conv.backward(grad\_out)
optimizer = new SGD()
conv.update\_parameters(optimizer, learning\_rate=0.01)
\end{verbatim}

\section{Notes on Dilation}
\begin{itemize}
    \item Dilation expands the receptive field, spacing kernel elements by \(d\_h, d\_w\).
    \item $ih = oh \cdot s\_h + kh \cdot d\_h$, $iw = ow \cdot s\_w + kw \cdot d\_w$ in \texttt{im2col}.
    \item Effective kernel size: $d\_h \cdot (k\_h - 1) + 1$, $d\_w \cdot (k\_w - 1) + 1$.
\end{itemize}

\section{Error Handling}
\begin{itemize}
    \item Validate input dimensions and parameters.
    \item Check for non-integer or negative output dimensions.
    \item Ensure gradient shape matches.
\end{itemize}

\section{Optimization Tips}
\begin{itemize}
    \item Use efficient matrix libraries (e.g., BLAS, Eigen, NumPy).
    \item Cache intermediate results.
    \item Parallelize loops in \texttt{im2col} and \texttt{col2im\_add}.
    \item Consider GPUs for CUDA.
\end{itemize}
